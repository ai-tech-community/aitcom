**Step 2: Implement community thread signal detection**

Detect clusters of similar questions/discussions in community threads:

```typescript
/**
 * Find recent community threads that could inspire challenges.
 * Looks for: unanswered questions, popular discussions, recurring topics.
 */
export async function detectCommunitySignals(
  dayWindow = 14,
): Promise<ChallengeSignal[]> {
  const payload = await getPayloadClient();
  const since = new Date(Date.now() - dayWindow * 24 * 60 * 60 * 1000);

  // Fetch recent forum threads
  const { docs: threads } = await payload.find({
    collection: "forum-threads",
    where: {
      createdAt: { greater_than: since.toISOString() },
    },
    sort: "-createdAt",
    limit: 50,
    depth: 0,
  });

  // Fetch recent ideas
  const { docs: ideas } = await payload.find({
    collection: "ideas",
    where: {
      createdAt: { greater_than: since.toISOString() },
    },
    sort: "-votes",
    limit: 20,
    depth: 0,
  });

  const signals: ChallengeSignal[] = [];

  // Popular threads with many replies = community interest
  for (const thread of threads) {
    signals.push({
      type: "community-thread",
      reference: `thread:${thread.id}`,
      summary: `Thread "${thread.title}" — active community discussion`,
      relevanceScore: 0.5,
      rawData: { threadId: thread.id, title: thread.title },
    });
  }

  // Popular ideas = community wants
  for (const idea of ideas) {
    signals.push({
      type: "member-request",
      reference: `idea:${idea.id}`,
      summary: `Idea "${idea.title}" with community votes`,
      relevanceScore: 0.7,
      rawData: { ideaId: idea.id, title: idea.title },
    });
  }

  return signals;
}
```

**Step 3: Implement activity-based signal detection**

Look at what members are doing/struggling with:

```typescript
/**
 * Detect signals from recent activity patterns.
 * Looks for: many members doing similar things, common failure patterns.
 */
export async function detectActivitySignals(
  dayWindow = 14,
): Promise<ChallengeSignal[]> {
  const since = new Date(Date.now() - dayWindow * 24 * 60 * 60 * 1000);

  // Find most common actions in the last N days
  const topActions = await db
    .select({
      action: activityEvents.action,
      count: sql<number>`count(*)`.as("count"),
    })
    .from(activityEvents)
    .where(gte(activityEvents.createdAt, since))
    .groupBy(activityEvents.action)
    .orderBy(desc(sql`count(*)`))
    .limit(10);

  const signals: ChallengeSignal[] = [];

  for (const row of topActions) {
    signals.push({
      type: "community-thread",
      reference: `activity:${row.action}`,
      summary: `Activity "${row.action}" performed ${row.count} times recently`,
      relevanceScore: Math.min(row.count / 50, 1),
      rawData: { action: row.action, count: row.count },
    });
  }

  return signals;
}
```

**Step 4: Create the combined signal collector**

```typescript
/**
 * Collect all signals from all sources.
 * Returns signals sorted by relevance.
 */
export async function collectSignals(
  dayWindow = 14,
): Promise<ChallengeSignal[]> {
  const [communitySignals, activitySignals] = await Promise.all([
    detectCommunitySignals(dayWindow),
    detectActivitySignals(dayWindow),
  ]);

  const allSignals = [...communitySignals, ...activitySignals];

  // Sort by relevance
  allSignals.sort((a, b) => b.relevanceScore - a.relevanceScore);

  return allSignals;
}
```

**Step 2: Implement the generation function**

```typescript
const client = new Anthropic();

export async function generateChallenge(
  signals: ChallengeSignal[],
  existingChallengeTitles: string[],
): Promise<GeneratedChallenge> {
  const signalSummary = signals
    .slice(0, 10)
    .map((s) => `- [${s.type}] ${s.summary} (relevance: ${s.relevanceScore.toFixed(2)})`)
    .join("\n");

  const existingList = existingChallengeTitles.length > 0
    ? `\n\nExisting challenges (do NOT duplicate):\n${existingChallengeTitles.map((t) => `- ${t}`).join("\n")}`
    : "";

  const response = await client.messages.create({
    model: "claude-sonnet-4-6",
    max_tokens: 2000,
    system: SYSTEM_PROMPT,
    messages: [
      {
        role: "user",
        content: `Generate ONE challenge based on these community signals:\n\n${signalSummary}${existingList}\n\nPick the most impactful signal and create a challenge around it. Use type "weekly" for small challenges (3 objectives max), "monthly" for bigger ones (5 objectives max), "open-ended" for ongoing ones.\n\nRespond with a JSON object matching this schema:\n{\n  title: string,\n  slug: string (kebab-case),\n  description: string (2-3 paragraphs explaining the challenge),\n  type: "weekly" | "monthly" | "open-ended",\n  difficulty: "beginner" | "intermediate" | "advanced" | "expert",\n  collaborationModel: "solo-ai",\n  objectives: [{ description: string, verification: "self-report" | "platform-action" | "peer-review", action?: "thread.reply" | "thread.create" | "knowledge.share" | "idea.submitted" | "idea.voted", targetCount: number }],\n  tags: string[],\n  signalSource: { type: "${signals[0]?.type ?? "community-thread"}", reference: string, summary: string }\n}`,
      },
    ],
  });

  const text = response.content[0]?.type === "text" ? response.content[0].text : "";
  const parsed = JSON.parse(text) as GeneratedChallenge;

  // Ensure collaboration model is solo-ai for Phase 1
  parsed.collaborationModel = "solo-ai";

  return parsed;
}
```

```typescript
// src/server/challenge-engine/lexical.ts

type LexicalNode = { type: string; version: number; [k: string]: unknown };

export function text(t: string, format = 0): LexicalNode {
  return { type: "text", text: t, version: 1, format };
}

export function paragraph(...children: LexicalNode[]): LexicalNode {
  return {
    type: "paragraph",
    version: 1,
    format: "",
    indent: 0,
    direction: "ltr",
    children,
  };
}

export function heading(tag: string, ...children: LexicalNode[]): LexicalNode {
  return {
    type: "heading",
    tag,
    version: 1,
    format: "",
    indent: 0,
    direction: "ltr",
    children,
  };
}

export function bulletList(...items: string[]): LexicalNode {
  return {
    type: "list",
    version: 1,
    listType: "bullet",
    tag: "ul",
    start: 1,
    format: "",
    indent: 0,
    direction: "ltr",
    children: items.map((t, i) => ({
      type: "listitem",
      version: 1,
      value: i + 1,
      indent: 0,
      format: "",
      direction: "ltr",
      children: [text(t)],
    })),
  };
}

export function hr(): LexicalNode {
  return { type: "horizontalrule", version: 1 };
}

export function lexical(...children: LexicalNode[]) {
  return {
    root: {
      type: "root",
      version: 1,
      direction: "ltr" as const,
      format: "" as const,
      indent: 0,
      children,
    },
  };
}

/**
 * Convert a plain text description (paragraphs separated by double newlines)
 * into a Lexical rich text structure.
 */
export function plainTextToLexical(description: string) {
  const paragraphs = description
    .split(/\n\n+/)
    .filter(Boolean)
    .map((p) => paragraph(text(p.trim())));
  return lexical(...paragraphs);
}
```

```typescript
// src/server/challenge-engine/publish.ts

import { getPayloadClient } from "@/server/payload";
import type { GeneratedChallenge } from "./generate";
import { plainTextToLexical } from "./lexical";

/**
 * Publish a generated challenge to Payload CMS.
 * Returns the created challenge ID.
 */
export async function publishChallenge(
  generated: GeneratedChallenge,
  options: { status?: "draft" | "active" } = {},
): Promise<{ id: number; slug: string }> {
  const payload = await getPayloadClient();
  const { status = "draft" } = options;

  // Check for duplicate slug
  const existing = await payload.find({
    collection: "challenges",
    where: { slug: { equals: generated.slug } },
    limit: 1,
  });

  const slug = existing.docs.length > 0
    ? `${generated.slug}-${Date.now()}`
    : generated.slug;

  const challenge = await payload.create({
    collection: "challenges",
    data: {
      title: generated.title,
      slug,
      description: plainTextToLexical(generated.description),
      type: generated.type,
      status,
      difficulty: generated.difficulty,
      publishedBy: "member",
      creatorId: "system-ai",
      generatedBy: "ai",
      collaborationModel: generated.collaborationModel,
      objectives: generated.objectives.map((obj) => ({
        description: obj.description,
        verification: obj.verification,
        action: obj.action,
        targetCount: obj.targetCount,
      })),
      rewards: {
        xpReward: 0, // Auto-calculated by the system
      },
      tags: generated.tags,
      signalSource: generated.signalSource,
      maxParticipants: 0,
      rankingMode: "collaboration",
    },
  });

  return { id: challenge.id, slug };
}
```

## Task 6: Create the Challenge Engine Orchestrator

This ties signal detection, generation, and publishing together.

**Files:**
- Create: `src/server/challenge-engine/index.ts`

**Step 1: Implement the orchestrator**

```typescript
// src/server/challenge-engine/index.ts

export { collectSignals, type ChallengeSignal } from "./signals";
export { generateChallenge, type GeneratedChallenge } from "./generate";
export { publishChallenge } from "./publish";
export { plainTextToLexical, lexical, paragraph, text, heading, bulletList, hr } from "./lexical";

import { collectSignals } from "./signals";
import { generateChallenge } from "./generate";
import { publishChallenge } from "./publish";
import { getPayloadClient } from "@/server/payload";

export interface EngineResult {
  signals: number;
  generated: {
    title: string;
    slug: string;
    difficulty: string;
    type: string;
  } | null;
  published: { id: number; slug: string } | null;
  error?: string;
}

/**
 * Run the full challenge engine pipeline:
 * 1. Collect signals from community activity
 * 2. Generate a challenge using AI
 * 3. Optionally publish it to the platform
 *
 * @param autoPublish - If true, publishes as "active". If false, publishes as "draft" for admin review.
 */
export async function runChallengeEngine(
  options: { autoPublish?: boolean; dayWindow?: number } = {},
): Promise<EngineResult> {
  const { autoPublish = false, dayWindow = 14 } = options;

  // 1. Collect signals
  const signals = await collectSignals(dayWindow);

  if (signals.length === 0) {
    return { signals: 0, generated: null, published: null, error: "No signals found" };
  }

  // 2. Get existing challenge titles to avoid duplicates
  const payload = await getPayloadClient();
  const { docs: existingChallenges } = await payload.find({
    collection: "challenges",
    where: {
      status: { in: ["active", "draft"] },
    },
    limit: 100,
    depth: 0,
  });
  const existingTitles = existingChallenges.map((c) => c.title);

  // 3. Generate
  const generated = await generateChallenge(signals, existingTitles);

  // 4. Publish
  const published = await publishChallenge(generated, {
    status: autoPublish ? "active" : "draft",
  });

  return {
    signals: signals.length,
    generated: {
      title: generated.title,
      slug: generated.slug,
      difficulty: generated.difficulty,
      type: generated.type,
    },
    published,
  };
}
```

```typescript
// src/server/api/routers/challenge-engine.ts

import { z } from "zod";
import { createTRPCRouter, protectedProcedure } from "@/server/api/trpc";
import { TRPCError } from "@trpc/server";
import { collectSignals } from "@/server/challenge-engine/signals";
import { runChallengeEngine } from "@/server/challenge-engine";

export const challengeEngineRouter = createTRPCRouter({
  /**
   * Get current community signals that could inspire challenges.
   * Admin-only: used to preview what the engine sees.
   */
  getSignals: protectedProcedure
    .input(z.object({ dayWindow: z.number().min(1).max(90).default(14) }).optional())
    .query(async ({ ctx, input }) => {
      // TODO: Add admin role check when role system exists
      const signals = await collectSignals(input?.dayWindow ?? 14);
      return { signals, count: signals.length };
    }),

  /**
   * Generate a challenge from community signals.
   * Creates as draft by default for admin review.
   */
  generate: protectedProcedure
    .input(
      z.object({
        autoPublish: z.boolean().default(false),
        dayWindow: z.number().min(1).max(90).default(14),
      }).optional(),
    )
    .mutation(async ({ ctx, input }) => {
      // TODO: Add admin role check when role system exists
      const result = await runChallengeEngine({
        autoPublish: input?.autoPublish ?? false,
        dayWindow: input?.dayWindow ?? 14,
      });

      if (result.error) {
        throw new TRPCError({
          code: "INTERNAL_SERVER_ERROR",
          message: result.error,
        });
      }

      return result;
    }),
});
```
